#define RX2_PIN 2 #define TX2_PIN 3 Preferences preferences; int current_pwr_level = 12; void applyBluetoothPower(int pwr) { if (pwr == 24) esp_ble_tx_power_set(ESP_BLE_PWR_TYPE_ADV, ESP_PWR_LVL_N24); else if (pwr == 12) esp_ble_tx_power_set(ESP_BLE_PWR_TYPE_ADV, ESP_PWR_LVL_N12); else if (pwr == 9) esp_ble_tx_power_set(ESP_BLE_PWR_TYPE_ADV, ESP_PWR_LVL_N9); else esp_ble_tx_power_set(ESP_BLE_PWR_TYPE_ADV, ESP_PWR_LVL_N12); } // 藍牙通道 1：處理手機貼緊轉發 485 密文 class OgtBridgeCallbacks: public BLECharacteristicCallbacks { void onWrite(BLECharacteristic *pChar) { String encryptedToken = pChar->getValue().c_str(); encryptedToken.trim(); if (encryptedToken.length() > 0) { SerialCOM2.println(encryptedToken); delay(200); ESP.restart(); } } }; // 藍牙通道 2：處理手機網頁直接透過藍牙修改功率 class OgtPwrCallbacks: public BLECharacteristicCallbacks { void onWrite(BLECharacteristic *pChar) { String cmd = pChar->getValue().c_str(); cmd.trim(); if (cmd.startsWith("SET_PWR:")) { int new_pwr = cmd.substring(8).toInt(); preferences.begin("ogt_config", false); preferences.putInt("pwr_level", new_pwr); preferences.end(); Serial.print("\n[藍牙指令] 成功透過網頁藍牙將功率改為: -"); Serial.print(new_pwr); Serial.println(" dBm"); delay(500); ESP.restart(); // 重啟套用 } } }; void setup() { Serial.begin(115200); delay(1000); Serial.println("\n=== OGT 智能門禁 [有線+藍牙雙向配置版] 啟動 ==="); SerialCOM2.begin(115200, SERIAL_8N1, RX2_PIN, TX2_PIN); preferences.begin("ogt_config", false); current_pwr_level = preferences.getInt("pwr_level", 12); applyBluetoothPower(current_pwr_level); BLEDevice::init("_Gate"); BLEServer *pServer = BLEDevice::createServer(); BLEService *pService = pServer->createService(SERVICE_UUID); // 綁定開門通道 BLECharacteristic *pCharKey = pService->createCharacteristic(CHARACTERISTIC_UUID_KEY, BLECharacteristic::PROPERTY_WRITE); pCharKey->setCallbacks(new OgtBridgeCallbacks()); // 綁定功率通道 BLECharacteristic *pCharPwr = pService->createCharacteristic(CHARACTERISTIC_UUID_PWR, BLECharacteristic::PROPERTY_WRITE); pCharPwr->setCallbacks(new OgtPwrCallbacks()); pService->start(); BLEAdvertising *pAdvertising = BLEDevice::getAdvertising(); pAdvertising->addServiceUUID(SERVICE_UUID); pAdvertising->setScanResponse(true); pAdvertising->setMinInterval(80); pAdvertising->setMaxInterval(80); pAdvertising->start(); Serial.print("[系統] 當前功率：-"); Serial.print(current_pwr_level); Serial.println(" dBm。雙向監聽中..."); } void loop() { // 485 有線網絡改功率依舊保留 if (SerialCOM2.available() > 0) { String cmd = SerialCOM2.readStringUntil('\n'); cmd.trim(); if (cmd.startsWith("SET_PWR:")) { int new_pwr = cmd.substring(8).toInt(); preferences.begin("ogt_config", false); preferences.putInt("pwr_level", new_pwr); preferences.end(); delay(500); ESP.restart(); } } delay(100); }
